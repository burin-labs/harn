  let q = sql("""
select *
from users
where id = {user_id}
""", {user_id: user_id})
  let named = named_sql("users_by_id", "one", "select * from users where id = {user_id}", {user_id: user_id})
  let raw = sql(r"select * from users where name = {name}", {name: "Ada"})
}
