fn divide(value: int) {
  const direct = value
    / 2
  const commented = value
    // keep the division on the following line
    / 2
  const nested_comment = value
    /* note /* nested */ end */
    / 2
  const modulo = value
    % 2
  // this begins a separate statement
  const separate = 3
  return [direct, commented, nested_comment, modulo, separate]
}

fn contextual(scope: int, block: int) {
  scope += 1
  block = block + scope
  scope(scope)
  block(block)
  const data = {scope: scope, block: block}
  return data.scope + data.block
}
fn scopes() {
  scope { const local = 1 }
  block { const local = 2 }
}

fn strings() {
  const single = r#"quoted "body" ${literal} \\ path"#
  const nested = r###"contains "# and "##, then ends"###
  const empty = r#""#
  return [single, nested, empty]
}
