fn negated() -> bool {
  return if true {
    const found = false
    !found
  } else { false }
}

fn commented() -> bool {
  return if true {
    const found = false
    // This starts another expression.
    !found
  } else { false }
}

  const value = 1
    // This continues the preceding expression.
    != 2

fn typed_closure() -> string {
  const render = { args: list<string> -> args.join(",") }
  const spaced = { args: list <string> -> args.join(",") }
  return render(["a"]) + spaced(["b"])
}
